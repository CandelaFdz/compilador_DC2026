
/*shadowing*/
function main() {
    int x, y;
    x = 10;
    y = 0;

    print("Valor inicial de x en main:", x);

    while (y < 3) {
        y = y + 1;
        print("Iteracion while:", y);
    }

    if (x > 0) {
        int x;
        x = 99;
        print("Valor local de x en el if:", x);
    }

    print("Valor de x en main tras salir del if:", x);
    return 0;
}
