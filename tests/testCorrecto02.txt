function calcularTres(int p1, int p2, int p3) {
    int cuenta;
    cuenta = (p1 + p2) * p3;
    return cuenta;
}

function main() {
    int a, b;
    a = 10;
    b = -4;

    /* Caso: 'resultado' se usa ANTES de su declaracion 'int' */
    resultado = calcularTres(a, b, 3) + 10 / 2;
    int resultado;

    print("=== CASO 1: ARITMETICA Y FUNCION DE 3 PARAMETROS ===");
    print("Resultado final (esperado 23):", resultado);
    print(resultado + 7);

    return 0;
}
