function sumar(int x, int y) {
    int resultado;
    resultado = x + y;
    return resultado;
}

function main() {
    int a, b;
    a = 10;
    b = 5 * 2;
    
    if (a == b) {
        print("Son iguales", a);
        a = sumar(a, b);
    } else {
        print("Son distintos");
    }
    
    while (a > 0) {
        a = a - 1;
    }
}
